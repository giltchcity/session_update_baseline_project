void require(bool ok,const char* msg){if(!ok)throw std::runtime_error(msg);}
FrameData frame(const hydra::Sensor::ConstPtr& sensor,TimeStamp stamp){
  hydra::InputData input(sensor);input.timestamp_ns=stamp;
  input.world_T_body=Eigen::Isometry3d::Identity();
  input.range_image=cv::Mat(2,4,CV_32FC1,cv::Scalar(3.f));
  input.depth_image=cv::Mat(2,4,CV_32FC1,cv::Scalar(3.f));
  input.label_image=cv::Mat(2,4,CV_32SC1,cv::Scalar(42));
  input.color_image=cv::Mat(2,4,CV_8UC3,cv::Scalar(0,0,0));
  FrameData data(input);data.instance_image=cv::Mat(2,4,CV_32SC1,cv::Scalar(7));
  data.dynamic_image=cv::Mat::zeros(2,4,CV_32SC1);return data;
}
int main(){
 hydra::PipelineConfig pc;pc.label_space.total_labels=256;pc.label_space.object_labels.insert(42);
 hydra::GlobalInfo::init(pc);
 hydra::Camera::Config cc;cc.min_range=.1;cc.max_range=10.;cc.width=4;cc.height=2;
 cc.cx=1.;cc.cy=1.;cc.fx=1.;cc.fy=1.;cc.extrinsics=hydra::ParamSensorExtrinsics::Config();
 auto camera=std::make_shared<hydra::Camera>(cc,"joint_evidence_fixture");
 require(hydra::GlobalInfo::instance().setSensor(camera),"register camera");
 auto store=std::make_shared<PhysicalEvidenceStore>();
 require(store->ingest(frame(camera,10)) && store->ingest(frame(camera,20)),"ingest two distinct frames");
 RayVerificator::Config cfg;cfg.depth_tolerance=.3f;
 RayVerificator verifier(cfg);verifier.setPhysicalEvidenceStore(store);
 const auto snapshot=verifier.physicalEvidenceSnapshot();
 spark_dsg::Mesh surface(false,true,false,true);surface.resizeVertices(1);surface.setPos(0,Point(0,0,3));
 const BoundingBox box(Point::Ones(),Point::Zero());
 const auto first=verifier.countProjectedPhysicalSurface(7,surface,box,snapshot,.05f,10,10);
 const auto second=verifier.countProjectedPhysicalSurface(7,surface,box,snapshot,.05f,20,20);
 const auto whole=verifier.countProjectedPhysicalSurface(7,surface,box,snapshot,.05f,10,20);
 require(first.support_rays==1 && second.support_rays==1 && whole.support_rays==2,"unchanged actual counts");
 const auto p10=snapshot->project(10,Point(0,0,3));
 const auto p20=snapshot->project(20,Point(0,0,3));
 const size_t key10=*first.support_indices.begin(),key20=*second.support_indices.begin();
 const size_t key20_whole=*std::max_element(whole.support_indices.begin(),whole.support_indices.end());
 require(key10==key20,"existing local-index key collision between distinct windows");
 require(key20_whole!=key20,"same physical measurement has different local-window key");
 require(p10.pixel_index==p20.pixel_index,"same pixel distinct sensor times");
 std::set<std::pair<TimeStamp,uint32_t>> exact{{10,p10.pixel_index},{20,p20.pixel_index},{20,p20.pixel_index}};
 require(exact.size()==2,"timestamp/pixel retains distinct rays and deduplicates aliases");
 // The same stored endpoint has ONE unknown round-to-mm phase. The hypotheses
 // query different world points on that same pixel, not two independent ranges.
 const auto a=snapshot->project(20,Point(0,0,2.70025f));
 const auto c=snapshot->project(20,Point(0,0,2.70000f));
 require(a.pixel_index==c.pixel_index && a.endpoint.measured_depth_m==c.endpoint.measured_depth_m,"one raw source");
 const auto ma=projectedRelationProbabilities(a,7,.3f);
 const auto mc=projectedRelationProbabilities(c,7,.3f);
 require(ma.occluded==0 && mc.occluded==0,"only surface/through near positive matching boundary");
 const double both=std::min(ma.supported,mc.supported);
 const double a_only=ma.supported-both,c_only=mc.supported-both;
 const double neither=1.-std::max(ma.supported,mc.supported);
 require(c_only==0. && std::abs(both+a_only+c_only+neither-1.)<1e-12,"shared phase normalized joint law");
 require(std::abs(both+a_only-ma.supported)<1e-12 && std::abs(both+c_only-mc.supported)<1e-12,"matches actual projector marginals");
 require(ma.free*mc.supported>0.1,"independent product assigns positive mass to impossible combination");
 std::cout<<std::setprecision(17)
 <<"{\"counts\":["<<first.support_rays<<","<<second.support_rays<<","<<whole.support_rays
 <<"],\"local_keys\":["<<key10<<","<<key20<<","<<key20_whole
 <<"],\"exact_unique_sources\":"<<exact.size()
 <<",\"pixel\":"<<a.pixel_index<<",\"support_marginals\":["<<ma.supported<<","<<mc.supported
 <<"],\"joint_support_support\":"<<both<<",\"joint_support_free\":"<<a_only
 <<",\"joint_free_support\":"<<c_only<<",\"joint_free_free\":"<<neither
 <<",\"wrong_independent_free_support\":"<<ma.free*mc.supported<<"}\n";
 hydra::GlobalInfo::reset();
}
