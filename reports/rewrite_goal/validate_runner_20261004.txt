    for a, b in SNAP_MAP:

def save():

def stat(pid):
    try:
        a = Path('/proc/%d/stat' % pid).read_text().rsplit(')', 1)[1].split()
        return {'start': a[19], 'group': int(a[2])}
    except (OSError, IndexError):
        return None


def group_cpu(pgid):
    # CPU seconds (user + system) of every live process in the stage's process group.
    tick = os.sysconf('SC_CLK_TCK')
    total = 0.0
    for d in Path('/proc').iterdir():
        if not d.name.isdigit():
            continue
        try:
            a = (d / 'stat').read_text().rsplit(')', 1)[1].split()
        except (OSError, IndexError):
            continue
        if int(a[2]) == pgid:
            total += (int(a[11]) + int(a[12])) / tick
    return total


REST_DOMAIN = {}


def run(stage, cmd, env_extra=None, watch=None, cwd=P, retry=True):
    assert shutil.disk_usage('/mnt/d').free >= 20 * 2**30, 'host D reserve'
    env = os.environ.copy()
    for k in ['BASE1_BUILD_DIR', 'BASE1_CANONICAL_PREFIX', 'BASE1_KHRONOS_SETUP', 'SESSION_UPDATE_CANONICAL_ROOT',
              'SESSION_UPDATE_CANONICAL_PREFIX', 'SESSION_UPDATE_CANONICAL_BUILD']:
        env.pop(k, None)
    env.update(PYTHONDONTWRITEBYTECODE='1', ROS_DOMAIN_ID='71')
    if SNAP:
        env.update(BASE1_CANONICAL_PREFIX=str(SNAP / 'install'), BASE1_BUILD_DIR=str(SNAP / 'build'))
        cmd = [snap(a) for a in cmd]
        cwd = snap(str(cwd))
        env_extra = {k: snap(v) for k, v in (env_extra or {}).items()}
    if threading.current_thread() is threading.main_thread():
        env.update(REST_DOMAIN)
    if env_extra:
        env.update(env_extra)
    log = LOGS / (stage + '.txt')
    t0 = time.monotonic()
    with log.open('w') as f:
        child = subprocess.Popen(cmd, cwd=cwd, env=env, stdout=f, stderr=subprocess.STDOUT, start_new_session=True)
        tok = stat(child.pid)
        row = {'stage': stage, 'command': cmd, 'pid': child.pid, 'pgid': tok['group'] if tok else None,
               'proc_start_token': tok['start'] if tok else None, 'owned': True, 'log': str(log)}
        with lock:
            rec['stages'].append(row)
        save()
        print('START', stage, child.pid, flush=True)
        last, sizes = time.monotonic(), None
        cpu_mark = (time.monotonic(), group_cpu(child.pid))
        while child.poll() is None:
            time.sleep(5)
            if watch:
                files = [log] + [x for g in watch for x in ROOT.glob(g)]
                s = {str(x): x.stat().st_size for x in files if x.exists()}
                if s != sizes:
                    sizes, last = s, time.monotonic()
                # A transport stall leaves the processes waiting; a long silent computation (the
                # session-end re-integration logs nothing for ~200 s) keeps them busy. Idle means no
                # own log growth and under a tenth of one core over the window.
                now = time.monotonic()
                if now - cpu_mark[0] >= 300:
                    cpu_now = group_cpu(child.pid)
                    busy = cpu_now - cpu_mark[1] > 0.1 * (now - cpu_mark[0])
                    cpu_mark = (now, cpu_now)
                    if busy:
                        last = max(last, now - 150)
                if time.monotonic() - last > 300 and stat(child.pid) == tok:
                    row['stop_reason'] = 'own_logs_idle_300s'
                    os.killpg(child.pid, signal.SIGTERM)
                    try:
                        child.wait(timeout=10)
                    except subprocess.TimeoutExpired:
                        if stat(child.pid) == tok:
                            os.killpg(child.pid, signal.SIGKILL)
                    break
        row.update(exit_code=child.wait(), elapsed_seconds=round(time.monotonic() - t0, 3))
    save()
    print('DONE', stage, row['exit_code'], row['elapsed_seconds'], flush=True)
    if row['exit_code'] != 0 and retry and row.get('stop_reason') == 'own_logs_idle_300s':
        # A stalled transport (no own log growth for 300 s; Fast DDS 2026-10-04/05) is not a
        # result: the formal output was not published, so clean orphaned segments and run once more.
        subprocess.run(['bash', '-c', 'source /opt/ros/jazzy/setup.bash && fastdds shm clean'],
                       stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        return run(stage + '_retry', cmd, env_extra, watch, cwd, retry=False)
    if row['exit_code'] != 0:
        rec['status'] = 'failed:' + stage
        save()
        raise SystemExit(row['exit_code'])
    return row

# Pipelined evaluation (user 2026-10-06): a session's evaluation starts as soon as its own mapping is
# done and runs alongside the next mapping; still one mapper at a time. The threads are joined at the end.
def background(fn, *args):
    t = threading.Thread(target=fn, args=args)
    t.start()
    threads.append(t)


def run_steps(steps):
    for stage, cmd in steps:
        run(stage, cmd)

def real_chain():
    proto = json.loads((R / 'baseline_lock.json').read_text())['input_protocol']['real_abc']
    base = ex('baseline_real_a_execution_20261002.json')['command']
    a_root = ROOT / 'real_a'
    env = dict(proto.get('env', {}))
    def session_cmd(session, out, input_state=None):
        c = list(base)
        rgbd = {'a': 'session_a_20260809_204010_201_flat_rgbd_30hz_1080p', 'b': 'session_b_20260810_030502_620_flat_rgbd_30hz_1080p',
                'c': 'session_c_20260822_182703_387_flat_rgbd_30hz_1080p'}[session]
        rep = {'--run-dir': str(FT / 'datasets/local_ab/rgbd' / rgbd),
               '--semantic-dir': str(FT / 'datasets/local_ab/semantics' / ('session_' + session)),
               '--instance-dir': str(FT / 'datasets/local_ab/instance_labels' / ('session_' + session)),
               '--output-state': str(out), '--session-id': 'rewrite_goal_%s_real_%s' % (TAG, session),
               '--session-start-ns': str(proto['session_start_ns'][session])}
        for k, v in rep.items():
            c[c.index(k) + 1] = v
        if input_state:
            c += ['--input-state', str(input_state), '--world-transform',
                  str(FT / 'datasets/local_ab/alignment' / ('session_%s_to_session_a.txt' % session))]
        return c
    outs = {}
    prev = None
    sessions = os.environ.get('RUNNER_REAL_SESSIONS', 'abc')  # e.g. 'a': real A alone (level-1 check)
    for session in sessions:
        out = ROOT / ('real_' + session)
        if os.environ.get('RUNNER_RESUME') == '1' and (out / 'final.4dmap.zpk').exists():
            # Resume after a transport stall: a session this tag already published is kept.
            print('SKIP', 'real_%s_mapping' % session, flush=True)
        else:
            run('real_%s_mapping' % session, session_cmd(session, out, prev), env_extra=env,
                watch=['.real_%s.incomplete.*/state_control/logs/*.log' % session])
        outs[session] = out
        prev = out
        # This session's evaluation runs alongside the next session's mapping.
        c = ['/usr/bin/python3', str(FT / 'eval_fast/run.py'), '--map', str(outs[session] / 'final.4dmap.zpk'),
             '--session', session, '--name', 'rewrite_goal_' + TAG, '--mode', 'full',
             '--out', str(ROOT / ('real_%s_evaluation' % session))]
        if session != 'a':
            c += ['--prior', str(outs[{'b': 'a', 'c': 'b'}[session]] / 'final.4dmap.zpk')]
        background(run, 'real_%s_full_evaluation' % session, c)
    return []  # evaluations already started
