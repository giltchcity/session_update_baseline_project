FrameData frame(const hydra::Sensor::ConstPtr& camera, TimeStamp stamp) {
 hydra::InputData input(camera);input.timestamp_ns=stamp;input.world_T_body=Eigen::Isometry3d::Identity();
 input.range_image=cv::Mat(8,8,CV_32FC1,cv::Scalar(3.f));
 input.depth_image=cv::Mat(8,8,CV_32FC1,cv::Scalar(3.f));
 input.label_image=cv::Mat(8,8,CV_32SC1,cv::Scalar(42));
 input.color_image=cv::Mat(8,8,CV_8UC3,cv::Scalar(0,0,0));
 FrameData data(input);data.instance_image=cv::Mat(8,8,CV_32SC1,cv::Scalar(7));
 data.dynamic_image=cv::Mat::zeros(8,8,CV_32SC1);return data;
}
spark_dsg::Mesh surface(bool right) {
 spark_dsg::Mesh m(false,true,false,true);m.resizeVertices(32);size_t i=0;
 for(int v=0;v<8;++v)for(int u=(right?4:0);u<(right?8:4);++u)
  m.setPos(i++,Eigen::Vector3f((u-3.5f)/20.f,(v-3.5f)/20.f,1.f).normalized()*3.f);
 return m;
}

  auto c=verifier.countProjectedPhysicalSurface(7,mesh,box,snap,.05f,10,40);

  return c.reliable_in_view;
 };

}
