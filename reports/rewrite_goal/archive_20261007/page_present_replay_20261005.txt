# Offline replay of the observed-absence test (Page's statistic, eq. (7)) under present models
# (Python source kept as TXT).
# Usage: python3 page_present_replay_20261005.txt [--model legacy|variance] SESSION_LOG [SESSION_LOG ...]
# Session logs are given in chain order (A, then B reading A, ...); the population statistics are
# carried between them as saveAbsenceSensorStatistics / loadAbsenceSensorStatistics do.
# Each ABSENCE_LOOK record is one scored look: k of n judged reliable samples seen through, the
# logged log ratio, and first_since[s] (judged samples not judged since candidate s). The replay
# recomputes the present Beta from the looks learned before ("judge, then learn"), the look's
# log ratio, the Page candidates and the 99:1 commitment, and joins the same round's ray counts
# (STATE_EVIDENCE_WINDOW): a commitment closes a state only with contradiction > support.
# legacy must reproduce the logged ratios and commitments (self-check).
# variance: the scatter of the present model is the plain second moment of the looks instead of
# 1.4826 MAD (heavy tails kept), otherwise identical.
# mixture: the present density is (1 - eps) Beta_core + eps Beta_wide; Beta_core is the legacy Beta,
# Beta_wide has the same mean and variance max(v_core, s_w^2). eps and s_w come from EM on the
# within-object deviations of in-place looks (each look minus its object's mean of earlier looks,
# from three earlier looks on): zero-mean two-scale normal mixture, Jeffreys pseudo-count for eps;
# eps = 0 while fewer than three deviations exist or no wide scale is identified.
# studentt: hierarchical normal model of a look's share, f ~ N(mu_j, s_j^2), (mu_j, s_j^2) ~ NIG(m0, k0, a0, b0);
# the present density is the Student-t posterior predictive of the object (truncated to [0, 1]), its
# hyperparameters maximize the marginal likelihood of all objects' in-place looks learned so far
# (this session and the sessions before; empirical Bayes, no constant). While fewer than three
# objects carry two looks the hyperparameters are not identified and the legacy envelope stands in.
# Candidates of an online window that ended at an online commitment the replay did not make are
# approximated by the first count of the new window (an upper bound); such rows are flagged.
import json, math, re, sys
try:
    import numpy as np
    from scipy import optimize, stats, special
except ImportError:
    np = None
from collections import defaultdict

LOG99 = math.log(99.0)
args = sys.argv[1:]
model = 'legacy'
if '--model' in args:
    i = args.index('--model'); model = args[i + 1]; del args[i:i + 2]
# --gate posterior (P21 derivation 1): a look is learned as in place iff, with it, the test's own
# log posterior odds of absence are not positive (Page statistic 0; MAP at loss 1 of the learning
# decision), instead of the identity gate (own >= 3 and own > seen-through).
gate = 'identity'
if '--gate' in args:
    i = args.index('--gate'); gate = args[i + 1]; del args[i:i + 2]
# --weight n (module group 2, effective sample size): with a count likelihood (model bb) the look's
# ratio already stands for its n judged samples, so a candidate's weight is the share of those n
# judged for the first time since it (each sample speaks once), not the share of the reliable
# surface; the reliable-surface weight belongs to the one-share-per-look likelihood.
weight_base = 'reliable'
if '--weight' in args:
    i = args.index('--weight'); weight_base = args[i + 1]; del args[i:i + 2]
# --prior: the inherited prior of P14 (ABSENCE_COMMIT prior_log_odds) is added to page[0] while the
# first window lasts, as online (commit at max(logit q + page[0], cusum) > log 99).
# --range: the population part of the present model is taken from in-place looks of the same range bin
# (P37's 0.5 m bins; a look's range = median range of its judged samples), the global population when
# the bin holds fewer than 3 looks.
use_range = '--range' in args
if use_range:
    args.remove('--range')
use_prior = '--prior' in args
if use_prior:
    args.remove('--prior')


def lbeta(a, b):
    return math.lgamma(a) + math.lgamma(b) - math.lgamma(a + b)


def scatter(values, centre=None):
    if not values:
        return 1e-4
    if model == 'variance':
        c = sum(values) / len(values) if centre is None else centre
        return max(1e-4, sum((v - c) ** 2 for v in values) / len(values))
    c = sorted(values)[len(values) // 2] if centre is None else centre
    dev = sorted(abs(v - c) for v in values)
    mad = dev[len(dev) // 2]
    return max(1e-4, (1.4826 * mad) ** 2)


class Present:
    def __init__(self):
        self.pool_n = 0.0
        self.pool_sum = 0.0
        self.pool_dev = []
        self.loaded_var = -1.0
        self.own = defaultdict(list)
        self.devs = []
        self.hist = defaultdict(lambda: (0, 0.0))  # uncapped count and sum of in-place looks
        self.eps, self.sw2 = 0.0, 0.0
        self.loaded_eps, self.loaded_sw2 = 0.0, 0.0
        self.carried = []          # in-place look lists of earlier sessions' objects
        self.nig = None            # fitted (m0, k0, a0, b0)
        self.kn = None             # (k, n) of the look being scored (model bb)
        self.nig_dirty = True
        self.counts = []           # (key, k, n) of learned in-place looks (model bbml), all sessions
        self.bin_pool = defaultdict(list)   # range bin -> in-place shares (--range), all sessions
        self.cur_bin = None
        self.c_ml = None

    def fit_c(self):
        # maximum marginal likelihood concentration of the in-place count model: each learned look
        # k ~ BetaBinomial(n, m_j c, (1 - m_j) c), m_j the object's mean share of its learned looks
        # (Jeffreys-smoothed counts). c is the overdispersion: deff = (n + c) / (1 + c).
        if len(self.counts) < 3:
            self.c_ml = None
            return
        tot = defaultdict(lambda: [0.0, 0.0])
        for key, k, n in self.counts:
            tot[key][0] += k; tot[key][1] += n
        data = [((tot[key][0] + 0.5) / (tot[key][1] + 1.0), k, n) for key, k, n in self.counts]
        def ll(lc):
            c = math.exp(lc)
            return sum(lbeta(k + m * c, n - k + (1 - m) * c) - lbeta(m * c, (1 - m) * c) for m, k, n in data)
        lo, hi = math.log(1e-3), math.log(1e6)
        g = (math.sqrt(5) - 1) / 2
        a, b = hi - g * (hi - lo), lo + g * (hi - lo)
        fa, fb = ll(a), ll(b)
        while hi - lo > 1e-6:
            if fa > fb:
                hi, b, fb = b, a, fa; a = hi - g * (hi - lo); fa = ll(a)
            else:
                lo, a, fa = a, b, fb; b = lo + g * (hi - lo); fb = ll(b)
        self.c_ml = math.exp(0.5 * (lo + hi))

    def population(self):
        lists = [list(v) for v in self.own.values() if v] + [l for l in self.carried if l]
        return lists

    def fit_nig(self):
        lists = self.population()
        if sum(1 for l in lists if len(l) >= 2) < 3:
            self.nig = None
            return
        suff = [(len(l), sum(l) / len(l), sum((x - sum(l) / len(l)) ** 2 for x in l)) for l in lists]
        def nll(t):
            m0, k0, a0, b0 = t[0], math.exp(t[1]), math.exp(t[2]), math.exp(t[3])
            tot = 0.0
            for n, xb, ss in suff:
                kn = k0 + n; an = a0 + n / 2.0
                bn = b0 + 0.5 * ss + k0 * n * (xb - m0) ** 2 / (2.0 * kn)
                tot += (math.lgamma(an) - math.lgamma(a0) + a0 * math.log(b0) - an * math.log(bn)
                        + 0.5 * (math.log(k0) - math.log(kn)) - 0.5 * n * math.log(2 * math.pi))
            return -tot
        allx = [x for l in lists for x in l]
        mx = sum(allx) / len(allx); vx = max(1e-6, sum((x - mx) ** 2 for x in allx) / len(allx))
        start = self.nig_raw if getattr(self, 'nig_raw', None) is not None else [mx, 0.0, math.log(2.0), math.log(vx)]
        r = optimize.minimize(nll, start, method='Nelder-Mead', options={'xatol': 1e-6, 'fatol': 1e-8, 'maxiter': 4000})
        self.nig_raw = list(r.x)
        self.nig = (r.x[0], math.exp(r.x[1]), math.exp(r.x[2]), math.exp(r.x[3]))

    def studentt_log_density(self, key, f):
        if self.nig_dirty:
            self.fit_nig(); self.nig_dirty = False
        if self.nig is None:
            return None
        m0, k0, a0, b0 = self.nig
        x = self.own[key]
        n = len(x)
        xb = sum(x) / n if n else 0.0
        ss = sum((v - xb) ** 2 for v in x)
        kn = k0 + n; mn = (k0 * m0 + n * xb) / kn; an = a0 + n / 2.0
        bn = b0 + 0.5 * ss + (k0 * n * (xb - m0) ** 2 / (2.0 * kn) if n else 0.0)
        scale = math.sqrt(bn * (kn + 1) / (an * kn)); df = 2 * an
        xe = min(0.995, max(max(0.005, mn), f))
        mass = stats.t.cdf(1.0, df, mn, scale) - stats.t.cdf(0.0, df, mn, scale)
        return stats.t.logpdf(xe, df, mn, scale) - math.log(max(mass, 1e-300))

    def fit_mixture(self):
        d = self.devs
        if len(d) < 3:  # too few of this session's own: the fit carried over (or none)
            self.eps, self.sw2 = self.loaded_eps, self.loaded_sw2
            return
        sc2 = max(1e-4, (1.4826 * sorted(abs(x) for x in d)[len(d) // 2]) ** 2)
        sw2 = max(1e-4, sum(x * x for x in d) / len(d))
        if sw2 <= sc2:
            self.eps, self.sw2 = 0.0, 0.0
            return
        eps = 0.1
        for _ in range(200):
            w = []
            for x in d:
                lc = math.log1p(-eps) - 0.5 * math.log(sc2) - 0.5 * x * x / sc2
                lw = math.log(eps) - 0.5 * math.log(sw2) - 0.5 * x * x / sw2
                w.append(1.0 / (1.0 + math.exp(lc - lw)))
            sw = sum(w)
            new_eps = (sw + 0.5) / (len(d) + 1.0)
            sc2 = max(1e-4, sum((1 - wi) * x * x for wi, x in zip(w, d)) / max(1e-9, len(d) - sw))
            sw2 = max(sc2, sum(wi * x * x for wi, x in zip(w, d)) / max(1e-9, sw))
            done = abs(new_eps - eps) < 1e-9
            eps = new_eps
            if done:
                break
        self.eps, self.sw2 = (eps, sw2) if sw2 > sc2 else (0.0, 0.0)

    def next_session(self):
        self.carried += [list(v) for v in self.own.values() if v]
        self.nig_dirty = True
        gv = scatter(self.pool_dev) if len(self.pool_dev) >= 3 else self.loaded_var
        self.loaded_var = gv
        self.pool_dev = []
        self.own = defaultdict(list)
        self.loaded_eps, self.loaded_sw2 = self.eps, self.sw2
        self.devs = []
        self.hist = defaultdict(lambda: (0, 0.0))

    def log_ratio(self, key, f):
        own = self.own[key]
        v0 = scatter(self.pool_dev) if len(self.pool_dev) >= 3 else self.loaded_var
        m0 = self.pool_sum / self.pool_n if self.pool_n >= 3 else 0.0
        if use_range and self.cur_bin is not None and len(self.bin_pool[self.cur_bin]) >= 3:
            bp = self.bin_pool[self.cur_bin]
            m0 = sum(bp) / len(bp); v0 = scatter(bp)
        if not (self.pool_n >= 3 and v0 > 0):
            m0, v0 = 0.05, 0.09
        own_n = len(own)
        own_m = sum(own) / own_n if own_n else 0.0
        own_v = scatter(own, own_m) if own_n >= 3 else 0.0
        use_n = own_n if own_n >= 3 else 0.0
        kk = self.k_eb() if model == 'ebk' else 3.0
        if kk is None:
            kk, use_n = 1.0, 0.0  # not identified: the population alone (as for own_n < 3)
        m = (use_n * own_m + kk * m0) / (use_n + kk)
        v = max(v0, (use_n * (own_v + (own_m - m) ** 2) + kk * (v0 + (m0 - m) ** 2)) / (use_n + kk))
        mc = min(0.999, max(0.001, m))
        var = max(1e-4, (m * m + v) - m * m)
        c = max(2.0, mc * (1 - mc) / var - 1)
        a, b = mc * c + 1e-3, (1 - mc) * c + 1e-3
        x = min(0.995, max(max(0.005, mc), f))
        core = -lbeta(a, b) + (a - 1) * math.log(x) + (b - 1) * math.log(1 - x)
        if model == 'bbml' and self.kn is not None and self.c_ml is not None:
            k, n = self.kn
            a, b = mc * self.c_ml, (1 - mc) * self.c_ml
            ke = max(k, mc * n)
            lbb = lbeta(ke + a, n - ke + b) - lbeta(a, b)
            return -math.log(n + 1.0) - lbb - (math.lgamma(n + 1) - math.lgamma(ke + 1) - math.lgamma(n - ke + 1))
        if model == 'bb' and self.kn is not None:
            # finite counts (module group 2, effective sample size): the same present Beta as a
            # beta-binomial on the look's (k, n) against the uniform share's 1/(n + 1); one-sided as
            # online (a count below the present mean counts as the mean).
            k, n = self.kn
            ke = max(k, mc * n)
            lbb = lbeta(ke + a, n - ke + b) - lbeta(a, b)
            return -math.log(n + 1.0) - lbb - (math.lgamma(n + 1) - math.lgamma(ke + 1) - math.lgamma(n - ke + 1))
        if model == 'studentt':
            lt = self.studentt_log_density(key, f)
            return -core if lt is None else -lt
        if model != 'mixture' or self.eps <= 0.0:
            return -core
        vw = max(var, self.sw2)
        cw = max(2.0, mc * (1 - mc) / vw - 1)
        aw, bw = mc * cw + 1e-3, (1 - mc) * cw + 1e-3
        wide = -lbeta(aw, bw) + (aw - 1) * math.log(x) + (bw - 1) * math.log(1 - x)
        hi = max(core, wide)
        return -(hi + math.log((1 - self.eps) * math.exp(core - hi) + self.eps * math.exp(wide - hi)))

    def k_eb(self):
        # ebk: pseudo-look count of the population = within-object over between-object variance
        # (normal hierarchy shrinkage), both robust: within from the deviations of each object's
        # looks from its median (MAD), between from the objects' medians minus the within share.
        objs = [sorted(v) for v in self.own.values() if len(v) >= 3]
        if len(objs) < 3:
            return None
        dev = []
        meds = []
        for v in objs:
            med = v[len(v) // 2]
            meds.append(med)
            dev.extend(x - med for x in v)
        sw2 = scatter(dev, 0.0)
        tb = scatter(meds) - sw2 * sum(1.0 / len(v) for v in objs) / len(objs)
        if tb <= 0:
            return None
        return sw2 / tb

    def learn(self, key, f):
        if use_range and self.cur_bin is not None:
            self.bin_pool[self.cur_bin].append(f)
        if model == 'bbml' and self.kn is not None:
            self.counts.append((key, self.kn[0], self.kn[1]))
            self.fit_c()
        own = self.own[key]
        count, total = self.hist[key]
        if model == 'mixture' and count >= 3 and len(self.devs) < 4096:
            self.devs.append(f - total / count)
            self.fit_mixture()
        self.hist[key] = (count + 1, total + f)
        if len(own) < 256:
            own.append(f)
            self.nig_dirty = True
        if len(own) == 3 and len(self.pool_dev) < 4096:
            mean3 = sum(own) / 3
            self.pool_dev.append(mean3)
            self.pool_n += 1
            self.pool_sum += mean3


def read(path):
    rows, rays = [], {}
    priors = {}
    for line in open(path, errors='ignore'):
        m = re.search(r'\] (ABSENCE_LOOK|STATE_EVIDENCE_WINDOW|ABSENCE_COMMIT) (.*)', line)
        if not m:
            continue
        d = dict(kv.split('=', 1) for kv in m.group(2).split() if '=' in kv)
        if m.group(1) == 'ABSENCE_COMMIT':
            priors.setdefault((d['inst'], d['slot']), float(d['prior_log_odds']))
            continue
        if m.group(1) == 'STATE_EVIDENCE_WINDOW':
            rays[(d['inst'], d['through'])] = (int(d['support']), int(d['contradiction']))
        else:
            rows.append(d)
    return rows, rays, priors


present = Present()
report = {'model': model, 'sessions': []}
for si, path in enumerate(args):
    if si:
        present.next_session()
    rows, rays, priors = read(path)
    first_window = defaultdict(lambda: True)
    g = defaultdict(int)            # global look index per state
    g_on = defaultdict(int)         # global index where the online window starts
    g_re = defaultdict(int)         # global index where the replay window starts
    page = defaultdict(list)        # replay candidates g_re .. g
    approx = defaultdict(bool)
    max_dev, events = 0.0, []
    for d in rows:
        key = (d['inst'], d['slot'])
        k, n, rel = int(d['k']), int(d['n']), int(d['reliable'])
        f = k / n
        since = [int(x) for x in d['first_since'].split(',')]
        gi = g[key]
        if len(since) != gi - g_on[key] + 1 and len(since) != 1:
            # another record of the same identity interleaves (uid-keyed records since 7633d8b, logs
            # without the record key): treat as a restarted window (approximation, flagged)
            g_on[key] = gi - len(since) + 1; g_re[key] = max(g_re[key], g_on[key]); page[key] = page[key][-(gi - g_re[key]):] if gi > g_re[key] else []
            approx[key] = True
        if len(since) == 1:          # the online window starts here (first look or after a commitment)
            if events and gi > 0:
                for e in reversed(events):
                    if (e['inst'], e['slot']) == key:
                        e['online_commit'] = True
                        break
            g_on[key] = gi
        present.kn = (k, n)
        smp = [x.split(':') for x in d.get('samples', '').split(';') if x.count(':') == 3]
        rr = sorted(float(x[2]) for x in smp if float(x[1]) >= 0.5)
        present.cur_bin = int(rr[len(rr) // 2] / 0.5) if rr else None
        llr = present.log_ratio(key, f)
        max_dev = max(max_dev, abs(llr - float(d['llr'])))
        p = page[key]
        p.append(0.0)
        for j in range(len(p)):
            c = g_re[key] + j
            if c >= g_on[key]:
                w = since[c - g_on[key]]
            else:
                w = since[0]
                approx[key] = True
            p[j] += min(1.0, w / max(1, n if (weight_base == 'n' and model in ('bb', 'bbml')) else rel)) * llr
        stat = max(0.0, max(p))
        before = priors.get(key, 0.0) + p[0] if (use_prior and first_window[key] and g_re[key] == 0) else -1e300
        committed = max(before, stat) > LOG99
        sup, con = rays.get((d['inst'], d['stamp']), (0, 0))
        events.append({'inst': d['inst'], 'slot': d['slot'], 'stamp': d['stamp'], 'f': round(f, 3),
                       'n': n, 'reliable': rel, 'llr': round(llr, 3), 'stat': round(stat, 3),
                       'replay_commit': committed, 'online_commit': False, 'approximate': approx[key],
                       'support': sup, 'contradiction': con})
        g[key] = gi + 1
        if committed:
            first_window[key] = False
            page[key] = []
            g_re[key] = gi + 1
            approx[key] = False
        learn = (d['identified'] == '1') if gate == 'identity' else (not committed and stat <= 0.0)
        if learn:
            present.learn(key, f)
    # an online commitment at the last look of a state is visible only through the ray-confirmed
    # closure; it is left unmarked here
    diff = [e for e in events if e['replay_commit'] != e['online_commit']]
    closes = lambda e, flag: e[flag] and e['contradiction'] > e['support']
    report['sessions'].append({
        'log': path.split('/runs/')[-1], 'looks': len(rows),
        'max_llr_deviation_vs_logged': round(max_dev, 6),
        'online_commits': sum(e['online_commit'] for e in events),
        'replay_commits': sum(e['replay_commit'] for e in events),
        'closures_online': sum(closes(e, 'online_commit') for e in events),
        'closures_replay': sum(closes(e, 'replay_commit') for e in events),
        'commit_differences': diff[:120]})
print(json.dumps(report, indent=1, default=float))
