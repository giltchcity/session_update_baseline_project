# Log-only statistic (post-hoc check, selects nothing): seen-through share f of looks of GT-static states in
# L2_FINAL2, and per-sample seen-through rate by incidence |cos| (proxy for distance to the projected outline)
# and by range. Samples field: verdict:|cos|:range:delta (verdict 1 within band, 2 beyond = seen through).
import re, statistics as st, collections, math
R = '/home/jixian/Desktop/FT/session_update_baseline/runs/rewrite_goal_L2_FINAL2_20261006/'
AB = {2, 3, 7, 10, 11, 12, 13, 18, 19, 20}; BC = {2, 3, 7, 10, 12, 14, 19, 20}
def static(sess, inst, inh):
    if sess == 'real_a': return True
    if sess == 'real_b': return (inst not in AB) if inh else inst != 2
    if sess == 'real_c': return (inst not in BC) if inh else inst != 2
MINCOS = math.cos(math.radians(75))  # placeholder, replaced below from config if found
looks = []
for sess in ('real_a', 'real_b', 'real_c'):
    for line in open(R + sess + '/control/logs/khronos.log', errors='ignore'):
        m = re.search(r'\] ABSENCE_LOOK (.*)', line)
        if not m: continue
        body = m.group(1)
        d = dict(kv.split('=', 1) for kv in body.split() if '=' in kv and not kv.startswith('samples='))
        if not static(sess, int(d['inst']), d['inherited'] == '1'): continue
        sm = re.search(r'samples=(\S+)', body)
        smp = []
        if sm and sm.group(1) != '-':
            for x in sm.group(1).split(';'):
                p = x.split(':')
                if len(p) == 4: smp.append((int(p[0]), float(p[1]), float(p[2])))
        n = int(d['n']); k = int(d['k'])
        if n == 0: continue
        looks.append(dict(sess=sess, inst=int(d['inst']), f=k / n, n=n, smp=smp))
print('GT-static looks', len(looks), 'per session', collections.Counter(l['sess'] for l in looks))
fs = [l['f'] for l in looks]
print('f: mean %.3f median %.3f sd %.3f p90 %.3f' % (st.mean(fs), st.median(fs), st.pstdev(fs), sorted(fs)[int(0.9 * len(fs))]))
# per-sample seen-through rate by |cos| bin and range bin (all measured samples of these looks)
cb = [0, .1, .2, .3, .5, .7, .9, 1.01]; rb = [0, 1, 1.5, 2, 2.5, 3, 3.5, 10]
def binof(x, edges):
    for i in range(len(edges) - 1):
        if edges[i] <= x < edges[i + 1]: return i
    return len(edges) - 2
tab = collections.defaultdict(lambda: [0, 0])
for l in looks:
    for v, c, r in l['smp']:
        t = tab[(binof(c, cb), binof(r, rb))]; t[0] += 1; t[1] += (v == 2)
print('\nper-sample beyond-band rate by |cos| (rows) x range (cols); cell = rate (count)')
print('|cos|\\range ' + ' '.join('%5.1f-%-4.1f' % (rb[j], rb[j + 1]) for j in range(len(rb) - 1)) + '   all')
for i in range(len(cb) - 1):
    row = []; tot = [0, 0]
    for j in range(len(rb) - 1):
        n_, s_ = tab[(i, j)]; tot[0] += n_; tot[1] += s_
        row.append('%5.3f(%5d)' % (s_ / n_, n_) if n_ else '      -    ')
    print('%3.1f-%-4.2f ' % (cb[i], cb[i + 1]) + ' '.join(row) + '  %5.3f(%6d)' % (tot[1] / max(1, tot[0]), tot[0]))
colt = []
for j in range(len(rb) - 1):
    n_ = sum(tab[(i, j)][0] for i in range(len(cb) - 1)); s_ = sum(tab[(i, j)][1] for i in range(len(cb) - 1))
    colt.append('%5.3f(%5d)' % (s_ / n_, n_) if n_ else '      -    ')
print('all       ' + ' '.join(colt))
# look level: how much of the variance of f is explained by grouping looks by (a) grazing share, (b) median range
def eta2(groups):
    allv = [x for g in groups.values() for x in g]; mu = st.mean(allv)
    sst = sum((x - mu) ** 2 for x in allv); ssb = sum(len(g) * (st.mean(g) - mu) ** 2 for g in groups.values() if g)
    return ssb / sst if sst else 0.0
ga, gr, gs, gi = (collections.defaultdict(list) for _ in range(4))
for l in looks:
    if not l['smp']: continue
    graz = sum(1 for v, c, r in l['smp'] if c < 0.3) / len(l['smp'])
    rmed = st.median(r for v, c, r in l['smp'])
    ga[min(4, int(graz * 10))].append(l['f']); gr[binof(rmed, rb)].append(l['f'])
    gs[l['sess']].append(l['f']); gi[(l['sess'], l['inst'])].append(l['f'])
print('\nlook-level share of var(f) explained (correlation ratio eta^2):')
print('  by grazing share (|cos|<0.3) of the look: %.3f' % eta2(ga))
print('  by median range of the look:             %.3f' % eta2(gr))
print('  by session:                              %.3f' % eta2(gs))
print('  by object (session, inst):               %.3f' % eta2(gi))
for name, g, lab in (('grazing share', ga, lambda b: '%.1f-%.1f' % (b / 10, (b + 1) / 10 if b < 4 else 1)), ('median range', gr, lambda b: '%.1f-%.1f m' % (rb[b], rb[b + 1]))):
    print(' f by', name + ':', '; '.join('%s n=%d mean %.3f sd %.3f' % (lab(b), len(v), st.mean(v), st.pstdev(v)) for b, v in sorted(g.items())))
# composition model: expected f of a look from its samples' (|cos|, range) cells, with the facing rule of
# eq. (7) (min_cos = cos 60 deg = 0.5: a beyond-band reading is a seen-through verdict only when facing).
MC = 0.5
p = {key: (v[1] / v[0] if v[0] else 0.0) for key, v in tab.items()}
fa, fh = [], []
for l in looks:
    if not l['smp']: continue
    ek = sum(p[(binof(c, cb), binof(r, rb))] for v, c, r in l['smp'] if c >= MC)
    en = sum(1.0 if c >= MC else 1.0 - p[(binof(c, cb), binof(r, rb))] for v, c, r in l['smp'])
    fa.append(l['f']); fh.append(ek / en if en else 0.0)
mu = st.mean(fa); sst = sum((x - mu) ** 2 for x in fa); sse = sum((x - y) ** 2 for x, y in zip(fa, fh))
print('\ncomposition model (each look: cells of its own samples, pooled per-cell rates): R^2 = %.3f; sd(f) %.3f, sd(f_hat) %.3f' % (1 - sse / sst, st.pstdev(fa), st.pstdev(fh)))
big = sorted(((x - y), x, y) for x, y in zip(fa, fh))[-5:]
print('largest residuals f - f_hat:', [(round(a, 2), round(b, 2), round(c, 2)) for a, b, c in big])
