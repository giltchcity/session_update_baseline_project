using Cell=std::tuple<int64_t,int64_t,int64_t>;
Cell cellOf(const Point& p,float size){
 const auto c=(p/size).array().floor().cast<int64_t>().eval();
 return {c.x(),c.y(),c.z()};
}
FrameData observation(const hydra::Sensor::ConstPtr& camera,TimeStamp stamp,
                      const Points& points,const std::set<Cell>& selected,float size,bool blank){
 hydra::InputData input(camera);input.timestamp_ns=stamp;
 input.world_T_body=Eigen::Isometry3d::Identity();input.world_T_body.translation().z()=-3.;
 input.range_image=cv::Mat::zeros(12,100,CV_32FC1);
 input.depth_image=cv::Mat::zeros(12,100,CV_32FC1);
 input.label_image=cv::Mat(12,100,CV_32SC1,cv::Scalar(15));
 input.color_image=cv::Mat(12,100,CV_8UC3,cv::Scalar(0,0,0));
 for(const auto& p:points){
  if(blank || !selected.count(cellOf(p,size)))continue;
  const Point pc=p+Point(0,0,3);
  int u=-1,v=-1;require(camera->projectPointToImagePlane(pc,u,v),"point projects");
  require(u>=0&&u<100&&v>=0&&v<12,"point in image");
  input.range_image.at<float>(v,u)=pc.norm();input.depth_image.at<float>(v,u)=pc.z();
 }
 FrameData result(input);result.instance_image=cv::Mat(12,100,CV_32SC1,cv::Scalar(801));
 result.dynamic_image=cv::Mat::zeros(12,100,CV_32SC1);return result;
}
