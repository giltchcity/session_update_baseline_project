def lbeta(a, b):
    return math.lgamma(a) + math.lgamma(b) - math.lgamma(a + b)


def robust_var(values, centre=None):
    if not values:
        return P['vfloor']
    if centre is None:
        c = sorted(values)
        centre = c[len(c) // 2]
    dev = sorted(abs(v - centre) for v in values)
    mad = dev[len(dev) // 2]
    if 'vfloor_budget' in FORMS:
        # sampling variance of one round's share under the sampling budget, at the scatter's centre
        fl = centre * (1 - centre) / P['cap']
        return max(fl, (1.4826 * mad) ** 2) if fl > 0 else max(P['vfloor'], (1.4826 * mad) ** 2)
    if (1.4826 * mad) ** 2 < P['vfloor']:
        BIND['robust_var_floor'] += 1
    return max(P['vfloor'], (1.4826 * mad) ** 2)


class Population:
    def __init__(self):
        self.n, self.sum, self.dev, self.loaded = 0.0, 0.0, [], -1.0
        self.hist = defaultdict(lambda: [0, 0.0, []])  # record -> history_n, history_sum, looks

    def next_session(self):
        self.loaded = robust_var(self.dev) if len(self.dev) >= 3 else self.loaded
        self.dev = []
        self.hist = defaultdict(lambda: [0, 0.0, []])

    def k_eb(self):
        objs = [sorted(v[2]) for v in self.hist.values() if len(v[2]) >= 3]
        if len(objs) < 3:
            return None
        dev, meds = [], []
        for v in objs:
            med = v[len(v) // 2]
            meds.append(med)
            dev.extend(x - med for x in v)
        sw2 = robust_var(dev, 0.0)
        tb = robust_var(meds) - sw2 * sum(1.0 / len(v) for v in objs) / len(objs)
        return sw2 / tb if tb > 0 else None

    def log_ratio(self, rec, f, kn=None):
        return self.model(rec, f, kn)[0]

    def model(self, rec, f, kn=None):
        h = P['hist']
        v0 = robust_var(self.dev) if len(self.dev) >= h else self.loaded
        m0 = self.sum / self.n if self.n >= h else 0.0
        if not (self.n >= h and v0 > 0):
            m0, v0 = P['m_cold'], P['v_cold']
        own_n, own_sum, looks = self.hist[rec]
        own_m = own_sum / own_n if own_n > 0 else 0.0
        own_v = robust_var(looks, own_m) if own_n >= h else 0.0
        k = P['k']
        use_n = own_n if own_n >= h else 0.0
        if 'ebk' in FORMS:
            ke = self.k_eb()
            if ke is None:
                ke, use_n = 1.0, 0.0
            k = ke
        m = (use_n * own_m + k * m0) / (use_n + k)
        v = max(v0, (use_n * (own_v + (own_m - m) ** 2) + k * (v0 + (m0 - m) ** 2)) / (use_n + k))
        mc = min(P['mhi'], max(P['mlo'], m))
        if mc != m:
            BIND['mean_clamp'] += 1
        if (m * m + v) - m * m < P['vfloor']:
            BIND['beta_var_floor'] += 1
        var = max(mc * (1 - mc) / P['cap'] if 'vfloor_budget' in FORMS else P['vfloor'], (m * m + v) - m * m)
        if mc * (1 - mc) / var - 1 < P['cfloor']:
            BIND['concentration_floor'] += 1
        c = max(P['cfloor'], mc * (1 - mc) / var - 1)
        a, b = mc * c + P['eps'], (1 - mc) * c + P['eps']
        if 'jeffreys' in FORMS and kn is not None:
            x = max(mc, (kn[0] + 0.5) / (kn[1] + 1.0))
        else:
            if f > P['fhi'] or max(P['flo'], mc) > max(mc, 0) and f < P['flo'] and mc < P['flo']:
                BIND['f_clamp'] += 1
            x = min(P['fhi'], max(max(P['flo'], mc), f))
        llr = -(-lbeta(a, b) + (a - 1) * math.log(x) + (b - 1) * math.log(1 - x))
        self.last = dict(own_n=own_n, own_m=round(own_m, 4), own_v=round(own_v, 6), m0=round(m0, 4), v0=round(v0, 6),
                         m=round(m, 4), v=round(var, 6), c=round(c, 2), x=round(x, 3), llr=round(llr, 3))
        return llr, mc, var

    def learn(self, rec, f):
        hs = self.hist[rec]
        hs[0] += 1
        hs[1] += f
        if len(hs[2]) < P['looks_cap']:
            hs[2].append(f)
        elif True:
            BIND['looks_cap'] += 1
        if hs[0] == 3 and len(self.dev) >= P['pool_cap']:
            BIND['pool_cap'] += 1
        if hs[0] == 3 and len(self.dev) < P['pool_cap']:
            self.dev.append(hs[1] / 3)
            self.n += 1
            self.sum += hs[1] / 3


def parse(line):
    m = re.search(r'\] (ABSENCE_LOOK|ABSENCE_UNSCORED|STATE_EVIDENCE_WINDOW|ABSENCE_COMMIT) (.*)', line)
    if not m:
        return None, None
    return m.group(1), dict(kv.split('=', 1) for kv in m.group(2).split() if '=' in kv)

def id_llr(rec, g):
    # mirror of the seen-through share: only an identity share below the in-place mean speaks for absence
    h = P['hist']

    m0 = pop_g.sum / pop_g.n if pop_g.n >= h else 0.0
    if not (pop_g.n >= h and v0 > 0):
        m0, v0 = (0.5, 1.0 / 12.0) if 'idbound' in FORMS else (1.0 - P['m_cold'], P['v_cold'])
    own_n, own_sum, looks_ = pop_g.hist[rec]
    own_m = own_sum / own_n if own_n > 0 else 0.0
    own_v = robust_var(looks_, own_m) if own_n >= h else 0.0
    use_n = own_n if own_n >= h else 0.0
    k = P['k']
    m = (use_n * own_m + k * m0) / (use_n + k)
    v = max(v0, (use_n * (own_v + (own_m - m) ** 2) + k * (v0 + (m0 - m) ** 2)) / (use_n + k))
    mc = min(P['mhi'], max(P['mlo'], m))
    var = max(P['vfloor'], v)
    c = max(P['cfloor'], mc * (1 - mc) / var - 1)
    a, b = mc * c + P['eps'], (1 - mc) * c + P['eps']
    if 'idbound' in FORMS:
        x = max(P['flo'], min(P['fhi'], g))  # two-sided
        logr = -lbeta(a, b) + (a - 1) * math.log(x) + (b - 1) * math.log(1 - x)  # log L_P - log L_A (A uniform)
        r = math.exp(min(50.0, logr))
        return math.log((1 - P['pf']) + P['pf'] * r) - math.log((1 - P['pm']) * r + P['pm'])
    x = max(P['flo'], min(min(P['fhi'], mc), g))
    return -(-lbeta(a, b) + (a - 1) * math.log(x) + (b - 1) * math.log(1 - x))

def beta_of(m, v):
    mc = min(P['mhi'], max(P['mlo'], m)); var = max(P['vfloor'], v)
    c = max(P['cfloor'], mc * (1 - mc) / var - 1)
    return mc * c + P['eps'], (1 - mc) * c + P['eps']
def lse2(ca, la, cb, lb_):
    t = [math.log(ca) + la] if ca > 0 else []
    if cb > 0: t.append(math.log(cb) + lb_)
    mx = max(t); return mx + math.log(sum(math.exp(x - mx) for x in t))
JL = {}
# Look-level confusion of the label channel (Dawid & Skene 1979, decision-directed / classification EM):
# y = labels favour absence (log r < 0). Present looks: scored looks of states not in a closed window
# (provisional); absent looks: looks of closed windows. Prior: the cold-start 0.05 as kRobustLooks pseudo-looks.
CONF = dict(pn=0.0, pk=0.0, an=0.0, ak=0.0)
CONF2 = dict(pn=0.0, pk=0.0, an=0.0, ak=0.0)
def conf_rates():
    if P['learnconf'] == 2:
        # Laplace's rule of succession; rho from depth-favoured in-place looks, rho_F from closed windows
        return (CONF2['pk'] + 1) / (CONF2['pn'] + 2), (CONF2['ak'] + 1) / (CONF2['an'] + 2)
    if not P['learnconf']:
        return P['pm'], P['pf']
    k = P['k']
    return (CONF['pk'] + k * 0.05) / (CONF['pn'] + k), (CONF['ak'] + k * 0.05) / (CONF['an'] + k)
def joint_llr(rec, g):
    pm_ = label_moments_P(rec)
    JL.clear()
    if pm_ is None or not ABS_POOL:
        return 0.0
    mP, vP = pm_
    nA = len(ABS_POOL); mabs = sum(ABS_POOL) / nA; vabs = robust_var(ABS_POOL, mabs)
    k = P['k']
    mA = (nA * mabs + k * mP) / (nA + k)
    vA = max(P['vfloor'], (nA * (vabs + (mabs - mA) ** 2) + k * (vP + (mP - mA) ** 2)) / (nA + k))
    aP, bP = beta_of(mP, vP); aA, bA = beta_of(mA, vA)
    JL.update(mP=round(mP, 3), mA=round(mA, 3), nA=nA)
    if (aP, bP) == (aA, bA):
        return 0.0
    x = max(P['flo'], min(P['fhi'], g))
    logr = (-lbeta(aP, bP) + (aP - 1) * math.log(x) + (bP - 1) * math.log(1 - x)) - \
           (-lbeta(aA, bA) + (aA - 1) * math.log(x) + (bA - 1) * math.log(1 - x))
    JL['logr'] = round(logr, 2)
    JL['logr_raw'] = logr
    pm, pf = conf_rates()
    JL['pm'], JL['pf'] = round(pm, 3), round(pf, 3)
    return lse2(1 - pf, 0.0, pf, logr) - lse2(1 - pm, logr, pm, 0.0)

def logbeta_pdf(x, ab):
    a, b = ab
    return -lbeta(a, b) + (a - 1) * math.log(x) + (b - 1) * math.log(1 - x)
def label_llr3(rec, shares, tots, rel):
    """shares[d] = (own share, foreign share); returns (llr_lab, logr, info)"""
    logr = 0.0; info = []
    for d in range(3):
        if tots[d] <= 0 or tots[d] < min(P['nmin'], rel):
            continue
        pm_ = node_P(d, rec); am_ = node_A(d)
        if pm_ is None or am_ is None:
            continue
        bp, ba = beta_of(*pm_), beta_of(*am_)
        if bp == ba:
            continue
        x = max(P['flo'], min(P['fhi'], shares[d][0]))
        lr = logbeta_pdf(x, bp) - logbeta_pdf(x, ba)
        logr += lr; info.append((d, round(shares[d][0], 3), round(pm_[0], 3), round(am_[0], 3), round(lr, 2)))
    if logr == 0.0:
        return 0.0, 0.0, info
    rho = (C3['pk'] + 1) / (C3['pn'] + 2); rhoF = (C3['ak'] + 1) / (C3['an'] + 2)
    return lse2(1 - rhoF, 0.0, rhoF, logr) - lse2(1 - rho, logr, rho, 0.0), logr, info + [('rho', round(rho, 3), round(rhoF, 3))]
def kish_weight():
    if not P['kish']:
        return 1.0
    xs, ys = AC['xs'], AC['ys']
    if len(xs) < P['hist']:
        return 1.0
    n = len(xs); mx = sum(xs) / n; my = sum(ys) / n
    sxy = sum((a - mx) * (b - my) for a, b in zip(xs, ys)); sxx = sum((a - mx) ** 2 for a in xs); syy = sum((b - my) ** 2 for b in ys)
    r1 = sxy / math.sqrt(sxx * syy) if sxx > 0 and syy > 0 else 0.0
    r1 = min(0.99, max(0.0, r1))
    return (1 - r1) / (1 + r1)   # 1 / deff, AR(1) effective-sample factor (Bayley & Hammersley 1946)
def parse_joint(d):
